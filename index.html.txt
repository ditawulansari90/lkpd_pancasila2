        // LINK GOOGLE APPS SCRIPT REKAP GURU
        const SCRIPT_URL = "https://script.google.com/macros/s/AKfycbzXZQcWGKAYLbz9X9O2_-Qs4mIlvtf9M0xlMa-0ZwxuRBmoE9dhjmXDyGjEUAxMpEHe/exec";

        // Web Audio API untuk efek suara
        const AudioContext = window.AudioContext || window.webkitAudioContext;
        let audioCtx;
        let totalSkorAkhir = 0;

        function initAudio() {
            if (!audioCtx) {
                audioCtx = new AudioContext();
            }
        }

        function playClickSound() {
            initAudio();
            if (!audioCtx) return;
            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            osc.type = 'sine';
            osc.frequency.setValueAtTime(400, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(800, audioCtx.currentTime + 0.08);
            gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.08);
            osc.connect(gain);
            gain.connect(audioCtx.destination);
            osc.start();
            osc.stop(audioCtx.currentTime + 0.08);
        }

        function playSuccessSound() {
            initAudio();
            if (!audioCtx) return;
            const now = audioCtx.currentTime;
            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            osc.type = 'triangle';
            osc.frequency.setValueAtTime(523.25, now);
            osc.frequency.setValueAtTime(659.25, now + 0.1);
            osc.frequency.setValueAtTime(783.99, now + 0.2);
            osc.frequency.setValueAtTime(1046.50, now + 0.3);
            gain.gain.setValueAtTime(0.2, now);
            gain.gain.exponentialRampToValueAtTime(0.01, now + 0.5);
            osc.connect(gain);
            gain.connect(audioCtx.destination);
            osc.start(now);
            osc.stop(now + 0.5);
        }

        function hitungNilai() {
            const nama = document.getElementById('inputNama').value;
            if (!nama) {
                alert('Silakan pilih nama lengkapmu terlebih dahulu!');
                document.getElementById('inputNama').focus();
                return;
            }

            let totalSkor = 0;

            // Bagian 1
            const b1Checks = document.querySelectorAll('.b1-check');
            let b1Count = 0;
            b1Checks.forEach(ch => {
                if (ch.checked) b1Count++;
            });
            totalSkor += (b1Count * 4);

            // Bagian 2
            for (let i = 1; i <= 6; i++) {
                if (b2Answers[i] === b2Kunci[i]) {
                    totalSkor += 5;
                }
            }

            // Bagian 3
            const b3Selects = document.querySelectorAll('.b3-select');
            b3Selects.forEach((sel, idx) => {
                if (sel.value === b3Kunci[idx]) {
                    totalSkor += 5;
                }
            });

            // Bagian 4
            let skorB4 = 0;
            const ans1 = document.getElementById('b4_soal1').value.trim();
            const ans2 = document.getElementById('b4_soal2').value.trim();
            if (ans1.length > 5) skorB4 += 12.5;
            if (ans2.length > 5) skorB4 += 12.5;
            totalSkor += Math.round(skorB4);

            totalSkorAkhir = totalSkor;

            // Tampilkan Hasil
            document.getElementById('totalSkor').innerText = totalSkorAkhir;
            const kelas = document.getElementById('inputKelas').value;
            const absen = document.getElementById('inputAbsen').value;
            document.getElementById('ringkasanSiswa').innerText = `Nama: ${nama} | Kelas: ${kelas} | No. Absen: ${absen || '-'}`;

            let pesan = "";
            if (totalSkorAkhir >= 85) {
                pesan = "🌟 Luar Biasa! Kamu adalah Detektif Norma & Anak Hebat sejati!";
            } else if (totalSkorAkhir >= 70) {
                pesan = "👍 Bagus Sekali! Kamu sudah memahami hak, kewajiban, dan norma dengan baik.";
            } else {
                pesan = "💪 Tetap Semangat! Pelajari lagi materi norma dan hak kewajiban ya!";
            }
            document.getElementById('umpanBalik').innerText = pesan;

            const hasilSec = document.getElementById('hasilSection');
            hasilSec.classList.remove('hidden');
            hasilSec.scrollIntoView({ behavior: 'smooth' });

            playSuccessSound();
        }

        function kirimKeGuru() {
            const nama = document.getElementById('inputNama').value;
            const kelas = document.getElementById('inputKelas').value;
            const status = document.getElementById("statusKirim");
            const btnKirim = document.getElementById("btnKirimGuru");

            if (!nama) {
                alert("Pilih nama siswa terlebih dahulu!");
                return;
            }

            status.className = "font-bold text-sm text-blue-600";
            status.innerText = "Sedang mengirim data ke guru...";
            btnKirim.disabled = true;
            btnKirim.classList.add("opacity-50", "cursor-not-allowed");

            const payload = {
                namaSiswa: nama,
                kelas: kelas,
                mataPelajaran: "Pendidikan Pancasila",
                namaTugas: "LKPD Pancasila",
                nilai: totalSkorAkhir
            };

            fetch(SCRIPT_URL, {
                method: "POST",
                mode: "no-cors",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify(payload)
            })
            .then(() => {
                status.className = "font-bold text-sm text-emerald-600";
                status.innerText = "✅ Berhasil! Nilai sudah masuk ke rekap Google Sheets guru.";
                btnKirim.disabled = false;
                btnKirim.classList.remove("opacity-50", "cursor-not-allowed");
            })
            .catch(err => {
                status.className = "font-bold text-sm text-rose-600";
                status.innerText = "❌ Gagal mengirim. Periksa koneksi internet.";
                btnKirim.disabled = false;
                btnKirim.classList.remove("opacity-50", "cursor-not-allowed");
            });
        }
